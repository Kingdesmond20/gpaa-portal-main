
===== 0001_foundation.sql =====
-- Foundation phase: roles, profiles, and admin-created invitations.
-- Run this once against a fresh Supabase project (SQL Editor, or
-- `supabase db push` if you've linked the Supabase CLI). See
-- supabase/README.md for full setup steps, including bootstrapping the
-- first admin.

create extension if not exists "pgcrypto";

create type public.user_role as enum ('member', 'admin');

-- One row per auth.users row, created automatically by the trigger below.
create table public.profiles (
  id uuid primary key references auth.users (id) on delete cascade,
  email text not null,
  full_name text,
  role public.user_role not null default 'member',
  created_at timestamptz not null default now()
);

alter table public.profiles enable row level security;

-- SECURITY DEFINER so RLS policies can call it without recursively
-- re-evaluating RLS on profiles (the standard Supabase pattern).
create or replace function public.is_admin()
returns boolean
language sql
security definer
set search_path = public
stable
as $$
  select exists (
    select 1 from public.profiles where id = auth.uid() and role = 'admin'
  );
$$;

create policy "Profiles are viewable by owner or admin"
  on public.profiles for select
  using (auth.uid() = id or public.is_admin());

create policy "Profiles are updatable by owner or admin"
  on public.profiles for update
  using (auth.uid() = id or public.is_admin());

grant select, update on public.profiles to authenticated;

-- Prevent a member from granting themselves admin via a direct table
-- update (the "owner" branch of the policy above only guards row
-- visibility/ownership, not which columns can change).
create or replace function public.prevent_role_self_escalation()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if new.role <> old.role and not public.is_admin() then
    raise exception 'Only admins can change roles';
  end if;
  return new;
end;
$$;

create trigger profiles_prevent_role_self_escalation
  before update on public.profiles
  for each row execute function public.prevent_role_self_escalation();

-- Populate profiles automatically on signup (including invite acceptance).
-- The role comes from user_metadata.role, set by the invite-user Edge
-- Function when an admin sends the invite; defaults to 'member' otherwise.
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  assigned_role public.user_role;
begin
  begin
    assigned_role := (new.raw_user_meta_data ->> 'role')::public.user_role;
  exception when others then
    assigned_role := null;
  end;
  if assigned_role is null then
    assigned_role := 'member';
  end if;

  insert into public.profiles (id, email, full_name, role)
  values (new.id, new.email, new.raw_user_meta_data ->> 'full_name', assigned_role);

  update public.invitations
  set status = 'accepted', accepted_at = now()
  where email = new.email and status = 'pending';

  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();

-- Admin-created invitations. Rows are normally written by the invite-user
-- Edge Function using the service-role key (bypasses RLS); the policies
-- below matter if an admin's own browser session ever reads/writes this
-- table directly.
create table public.invitations (
  id uuid primary key default gen_random_uuid(),
  email text not null,
  role public.user_role not null default 'member',
  status text not null default 'pending' check (status in ('pending', 'accepted', 'revoked')),
  invited_by uuid references public.profiles (id),
  created_at timestamptz not null default now(),
  accepted_at timestamptz
);

alter table public.invitations enable row level security;

create policy "Invitations are viewable by admins"
  on public.invitations for select
  using (public.is_admin());

create policy "Invitations are insertable by admins"
  on public.invitations for insert
  with check (public.is_admin());

create policy "Invitations are updatable by admins"
  on public.invitations for update
  using (public.is_admin());

grant select, insert, update on public.invitations to authenticated;

===== 0002_membership_applications.sql =====
-- Membership workflow phase: persist membership applications, and let the
-- handle_new_user() trigger from 0001 also link a new profile back to any
-- application submitted under that email before they had an account. Run
-- this after 0001_foundation.sql. See supabase/README.md.

create table public.membership_applications (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references public.profiles (id),
  category text not null,
  first_name text not null,
  last_name text not null,
  email text not null,
  phone text not null,
  gender text not null,
  date_of_birth date not null,
  mdc_license_number text not null,
  staff_id text not null,
  employer text not null,
  region text not null,
  district text not null,
  facility text not null,
  address text not null,
  status text not null default 'pending'
    check (status in ('pending', 'approved', 'rejected', 'needs_correction')),
  admin_notes text,
  correction_message text,
  reviewed_by uuid references public.profiles (id),
  reviewed_at timestamptz,
  submitted_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

alter table public.membership_applications enable row level security;

-- Public application form: anyone (including anon) can submit, but only as
-- a fresh 'pending' row with no review fields set â€” a client can't self-approve.
create policy "Anyone can submit a membership application"
  on public.membership_applications for insert
  with check (
    status = 'pending'
    and admin_notes is null
    and correction_message is null
    and reviewed_by is null
    and reviewed_at is null
  );

create policy "Applications are viewable by their owner or an admin"
  on public.membership_applications for select
  using (auth.uid() = user_id or public.is_admin());

-- No self-service edit in this phase (see plan's "assumption to flag") â€”
-- only admins update applications, via the review-application Edge
-- Function (using the service-role key) or directly in this phase.
create policy "Applications are updatable by admins"
  on public.membership_applications for update
  using (public.is_admin());

grant select, insert on public.membership_applications to anon;
grant select, insert, update on public.membership_applications to authenticated;

create or replace function public.touch_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

create trigger membership_applications_touch_updated_at
  before update on public.membership_applications
  for each row execute function public.touch_updated_at();

-- Redefine handle_new_user() (from 0001) to also backfill user_id onto any
-- application(s) submitted under this email before the applicant had an
-- account â€” e.g. they applied anonymously, got approved, and are now
-- accepting their invite.
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  assigned_role public.user_role;
begin
  begin
    assigned_role := (new.raw_user_meta_data ->> 'role')::public.user_role;
  exception when others then
    assigned_role := null;
  end;
  if assigned_role is null then
    assigned_role := 'member';
  end if;

  insert into public.profiles (id, email, full_name, role)
  values (new.id, new.email, new.raw_user_meta_data ->> 'full_name', assigned_role);

  update public.invitations
  set status = 'accepted', accepted_at = now()
  where email = new.email and status = 'pending';

  update public.membership_applications
  set user_id = new.id
  where email = new.email and user_id is null;

  return new;
end;
$$;

===== 0003_documents_and_payments.sql =====
-- Documents & Payments phase: an admin-curated document library with
-- public/members-only visibility, Paystack payment tracking, generated
-- membership certificates, and the profile columns that drive membership
-- standing (expiry date, reminder tracking). Run after 0002. See
-- supabase/README.md.

-- ---------------------------------------------------------------------------
-- Document library
-- ---------------------------------------------------------------------------

create table public.documents (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  description text not null,
  category text not null check (category in ('Financial', 'Reports', 'Calendar', 'Policies', 'Forms')),
  storage_path text not null unique,
  file_size_bytes bigint,
  visibility text not null default 'public' check (visibility in ('public', 'members')),
  uploaded_by uuid references public.profiles (id),
  created_at timestamptz not null default now()
);

alter table public.documents enable row level security;

-- 'members' visibility means "any authenticated user" (this org has no
-- finer-grained tiering) â€” public docs are visible to everyone including
-- logged-out visitors.
create policy "Documents are visible per their visibility"
  on public.documents for select
  using (visibility = 'public' or auth.role() = 'authenticated');

create policy "Admins manage documents"
  on public.documents for all
  using (public.is_admin())
  with check (public.is_admin());

grant select on public.documents to anon;
grant select, insert, update, delete on public.documents to authenticated;

insert into storage.buckets (id, name, public)
values ('documents', 'documents', false)
on conflict (id) do nothing;

create policy "Public document files are downloadable by anyone"
  on storage.objects for select
  using (
    bucket_id = 'documents'
    and exists (
      select 1 from public.documents d
      where d.storage_path = storage.objects.name and d.visibility = 'public'
    )
  );

create policy "Members-only document files are downloadable by authenticated users"
  on storage.objects for select
  using (
    bucket_id = 'documents'
    and auth.role() = 'authenticated'
    and exists (
      select 1 from public.documents d
      where d.storage_path = storage.objects.name and d.visibility = 'members'
    )
  );

create policy "Admins manage document files"
  on storage.objects for all
  using (bucket_id = 'documents' and public.is_admin())
  with check (bucket_id = 'documents' and public.is_admin());

-- ---------------------------------------------------------------------------
-- Payments (Paystack)
-- ---------------------------------------------------------------------------

create table public.payments (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references public.profiles (id),
  application_id uuid references public.membership_applications (id),
  category text not null,
  amount_pesewas integer not null,
  currency text not null default 'GHS',
  paystack_reference text not null unique,
  status text not null default 'pending' check (status in ('pending', 'success', 'failed')),
  period_start date,
  period_end date,
  created_at timestamptz not null default now(),
  paid_at timestamptz
);

alter table public.payments enable row level security;

create policy "Users view their own payments, admins any"
  on public.payments for select
  using (auth.uid() = user_id or public.is_admin());

-- No insert/update policy for anon/authenticated: rows are written only by
-- the create-payment / verify-payment Edge Functions via the service-role
-- key (which bypasses RLS), so a client can never fabricate a 'success'
-- payment or set its own amount.
grant select on public.payments to authenticated;

-- ---------------------------------------------------------------------------
-- Membership certificates
-- ---------------------------------------------------------------------------

create table public.membership_certificates (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references public.profiles (id),
  payment_id uuid references public.payments (id),
  certificate_number text not null unique,
  category text not null,
  issued_at timestamptz not null default now(),
  valid_from date not null,
  valid_until date not null,
  storage_path text not null
);

alter table public.membership_certificates enable row level security;

create policy "Users view their own certificates, admins any"
  on public.membership_certificates for select
  using (auth.uid() = user_id or public.is_admin());

grant select on public.membership_certificates to authenticated;

insert into storage.buckets (id, name, public)
values ('certificates', 'certificates', false)
on conflict (id) do nothing;

-- Certificate files are stored at "{user_id}/{certificate_id}.pdf".
create policy "Users download their own certificate files, admins any"
  on storage.objects for select
  using (
    bucket_id = 'certificates'
    and (public.is_admin() or (storage.foldername(name))[1] = auth.uid()::text)
  );

-- No client insert policy on the certificates bucket either â€” only the
-- verify-payment Edge Function (service role) writes certificate files.

-- ---------------------------------------------------------------------------
-- Membership standing on profiles
-- ---------------------------------------------------------------------------

alter table public.profiles
  add column membership_expiry_date date,
  add column expiry_reminder_sent_at timestamptz;

===== 0004_harden_role_assignment.sql =====
-- Security fix: stop trusting user-supplied metadata for the role.
--
-- handle_new_user() (0001, redefined in 0002) read the new user's role from
-- raw_user_meta_data.role. Supabase's public sign-up endpoint accepts
-- arbitrary metadata from anyone holding the anon key (which ships in the
-- site's JavaScript), so anyone could sign up with { role: 'admin' } and
-- become an admin.
--
-- The role now comes from the `invitations` table, which only admins (or the
-- service-role Edge Functions) can write. invite-user and review-application
-- insert the pending invitation BEFORE calling inviteUserByEmail, so the row
-- exists by the time this trigger fires. No pending invitation => 'member'.
--
-- Also fixes invitations never leaving 'pending': inviteUserByEmail creates
-- the auth.users row immediately, so marking the invite accepted in the
-- insert trigger happened at invite time. It now happens on first sign-in.

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  assigned_role public.user_role;
begin
  select role into assigned_role
  from public.invitations
  where lower(email) = lower(new.email) and status = 'pending'
  order by created_at desc
  limit 1;

  if assigned_role is null then
    assigned_role := 'member';
  end if;

  insert into public.profiles (id, email, full_name, role)
  values (new.id, new.email, new.raw_user_meta_data ->> 'full_name', assigned_role);

  -- Backfill user_id onto applications submitted before the account existed.
  update public.membership_applications
  set user_id = new.id
  where lower(email) = lower(new.email) and user_id is null;

  return new;
end;
$$;

create or replace function public.mark_invitation_accepted()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if old.last_sign_in_at is null and new.last_sign_in_at is not null then
    update public.invitations
    set status = 'accepted', accepted_at = now()
    where lower(email) = lower(new.email) and status = 'pending';
  end if;
  return new;
end;
$$;

drop trigger if exists on_auth_user_first_sign_in on auth.users;
create trigger on_auth_user_first_sign_in
  after update of last_sign_in_at on auth.users
  for each row execute function public.mark_invitation_accepted();

-- Backfill: invitations for people who have already signed in.
update public.invitations i
set status = 'accepted', accepted_at = coalesce(i.accepted_at, now())
from auth.users u
where lower(u.email) = lower(i.email)
  and u.last_sign_in_at is not null
  and i.status = 'pending';

-- Bootstrap fix: the README's first-admin step (`update public.profiles set
-- role = 'admin' ...` in the SQL Editor) was blocked by
-- prevent_role_self_escalation(): with no user JWT auth.uid() is null, so
-- is_admin() is false and the trigger raised. Only block role changes made
-- through a signed-in user's session; direct database access (SQL Editor,
-- service role) has auth.uid() null and is allowed. The `anon` role has no
-- UPDATE grant on profiles, so this doesn't open a path for the public.
create or replace function public.prevent_role_self_escalation()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if new.role <> old.role and auth.uid() is not null and not public.is_admin() then
    raise exception 'Only admins can change roles';
  end if;
  return new;
end;
$$;

===== 0005_tighten_policies_and_grants.sql =====
-- Found by the end-to-end test run against the live project.
--
-- 1. Application spoofing: the "Anyone can submit" insert policy (0002) never
--    checked user_id, so a signed-in user could file an application under
--    ANOTHER user's account (it would then show up on the victim's dashboard
--    and could be approved into their profile). Now an application is either
--    anonymous (user_id null) or owned by the caller.
drop policy if exists "Anyone can submit a membership application" on public.membership_applications;

create policy "Anyone can submit a membership application"
  on public.membership_applications for insert
  with check (
    status = 'pending'
    and admin_notes is null
    and correction_message is null
    and reviewed_by is null
    and reviewed_at is null
    and (user_id is null or user_id = auth.uid())
  );

-- 2. Defense in depth on table privileges. Supabase grants anon/authenticated
--    every privilege on new public tables by default, leaving RLS as the ONLY
--    guard. Revoke what no client flow needs, so a missing/loosened policy
--    can't silently become a write path.

-- Payments and certificates are written only by Edge Functions (service role).
revoke all on public.payments from anon, authenticated;
revoke all on public.membership_certificates from anon, authenticated;
grant select on public.payments to authenticated;
grant select on public.membership_certificates to authenticated;

-- Anonymous visitors need nothing on these.
revoke all on public.profiles from anon;
revoke all on public.invitations from anon;

-- Anonymous visitors may only submit applications and read public documents.
revoke all on public.membership_applications from anon;
grant insert on public.membership_applications to anon;
revoke all on public.documents from anon;
grant select on public.documents to anon;

-- TRUNCATE ignores RLS entirely; nobody on the client side needs it (or
-- REFERENCES/TRIGGER), and applications are never hard-deleted by clients.
revoke truncate, references, trigger on all tables in schema public from anon, authenticated;
revoke delete on public.membership_applications from authenticated;

===== 0006_awards.sql =====
-- Awards & public voting, built on Supabase (replaces the never-deployed
-- gpaa-voting-platform API as the backend for the /awards pages).
--
-- Model: campaign -> categories -> contestants (many-to-many with categories).
-- A vote is bought through an ORDER: quantity x unit price, priced on the
-- server, stored against a peppered phone hash (never the number), and counted
-- only once its payment is confirmed server-side. Orders are written solely by
-- the awards-vote Edge Function (service role); browsers can never insert,
-- update or mark an order paid.

-- ---------------------------------------------------------------------------
-- Campaigns
-- ---------------------------------------------------------------------------
create table public.award_campaigns (
  id uuid primary key default gen_random_uuid(),
  slug text not null unique check (slug ~ '^[a-z0-9]+(-[a-z0-9]+)*$'),
  name text not null,
  description text,
  status text not null default 'DRAFT'
    check (status in ('DRAFT', 'SCHEDULED', 'OPEN', 'PAUSED', 'CLOSED', 'ARCHIVED')),
  currency text not null default 'GHS' check (char_length(currency) = 3),
  starts_at timestamptz,
  ends_at timestamptz,
  timezone text not null default 'Africa/Accra',
  vote_unit_price numeric(12, 2) not null check (vote_unit_price > 0),
  max_quantity_per_transaction integer not null default 100 check (max_quantity_per_transaction > 0),
  max_transaction_amount numeric(12, 2) not null default 1000 check (max_transaction_amount > 0),
  -- Vote counts are hidden from the public unless an admin turns this on.
  show_results boolean not null default false,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  check (ends_at is null or starts_at is null or ends_at > starts_at)
);

create trigger award_campaigns_touch_updated_at
  before update on public.award_campaigns
  for each row execute function public.touch_updated_at();

-- ---------------------------------------------------------------------------
-- Categories, contestants, and their link table
-- ---------------------------------------------------------------------------
create table public.award_categories (
  id uuid primary key default gen_random_uuid(),
  campaign_id uuid not null references public.award_campaigns (id) on delete cascade,
  slug text not null check (slug ~ '^[a-z0-9]+(-[a-z0-9]+)*$'),
  name text not null,
  description text,
  display_order integer not null default 0,
  created_at timestamptz not null default now(),
  unique (campaign_id, slug)
);

create table public.award_contestants (
  id uuid primary key default gen_random_uuid(),
  campaign_id uuid not null references public.award_campaigns (id) on delete cascade,
  slug text not null check (slug ~ '^[a-z0-9]+(-[a-z0-9]+)*$'),
  full_name text not null,
  bio text,
  photo_url text,
  -- DISQUALIFIED contestants disappear from the public site and can't be voted for.
  status text not null default 'ACTIVE' check (status in ('ACTIVE', 'DISQUALIFIED')),
  created_at timestamptz not null default now(),
  unique (campaign_id, slug)
);

create table public.award_contestant_categories (
  contestant_id uuid not null references public.award_contestants (id) on delete cascade,
  category_id uuid not null references public.award_categories (id) on delete cascade,
  primary key (contestant_id, category_id)
);

create index award_categories_campaign_idx on public.award_categories (campaign_id, display_order);
create index award_contestants_campaign_idx on public.award_contestants (campaign_id);
create index award_contestant_categories_category_idx on public.award_contestant_categories (category_id);

-- ---------------------------------------------------------------------------
-- Vote orders (server-written only)
-- ---------------------------------------------------------------------------
create table public.award_vote_orders (
  id uuid primary key default gen_random_uuid(),
  campaign_id uuid not null references public.award_campaigns (id),
  category_id uuid not null references public.award_categories (id),
  contestant_id uuid not null references public.award_contestants (id),
  quantity integer not null check (quantity > 0),
  unit_price numeric(12, 2) not null check (unit_price > 0),
  amount numeric(12, 2) not null check (amount > 0),
  currency text not null,
  -- HMAC-SHA256(pepper, normalised phone). The number itself is never stored.
  phone_hash text not null,
  status text not null default 'PENDING_PAYMENT'
    check (status in ('PENDING_PAYMENT', 'PAYMENT_SUCCESS', 'FAILED')),
  payment_reference text not null unique,
  idempotency_key text,
  -- True for votes confirmed by the test-mode PAYMENT_MODE=simulated switch.
  is_simulated boolean not null default false,
  created_at timestamptz not null default now(),
  paid_at timestamptz,
  unique (campaign_id, idempotency_key)
);

create index award_vote_orders_tally_idx
  on public.award_vote_orders (campaign_id, contestant_id, category_id)
  where status = 'PAYMENT_SUCCESS';
create index award_vote_orders_phone_idx on public.award_vote_orders (phone_hash, created_at);

-- ---------------------------------------------------------------------------
-- Row-level security
-- ---------------------------------------------------------------------------
alter table public.award_campaigns enable row level security;
alter table public.award_categories enable row level security;
alter table public.award_contestants enable row level security;
alter table public.award_contestant_categories enable row level security;
alter table public.award_vote_orders enable row level security;

-- Public can read everything about a campaign once it is out of DRAFT and not
-- ARCHIVED. Admins can read and write all of it.
create policy "Published campaigns are public"
  on public.award_campaigns for select
  using (status in ('SCHEDULED', 'OPEN', 'PAUSED', 'CLOSED') or public.is_admin());
create policy "Admins manage campaigns"
  on public.award_campaigns for all
  using (public.is_admin()) with check (public.is_admin());

create policy "Categories of published campaigns are public"
  on public.award_categories for select
  using (
    public.is_admin()
    or exists (
      select 1 from public.award_campaigns c
      where c.id = award_categories.campaign_id
        and c.status in ('SCHEDULED', 'OPEN', 'PAUSED', 'CLOSED')
    )
  );
create policy "Admins manage categories"
  on public.award_categories for all
  using (public.is_admin()) with check (public.is_admin());

create policy "Active contestants of published campaigns are public"
  on public.award_contestants for select
  using (
    public.is_admin()
    or (
      status = 'ACTIVE'
      and exists (
        select 1 from public.award_campaigns c
        where c.id = award_contestants.campaign_id
          and c.status in ('SCHEDULED', 'OPEN', 'PAUSED', 'CLOSED')
      )
    )
  );
create policy "Admins manage contestants"
  on public.award_contestants for all
  using (public.is_admin()) with check (public.is_admin());

create policy "Contestant category links follow contestant visibility"
  on public.award_contestant_categories for select
  using (
    public.is_admin()
    or exists (
      select 1 from public.award_contestants ct
      where ct.id = award_contestant_categories.contestant_id
        and ct.status = 'ACTIVE'
        and exists (
          select 1 from public.award_campaigns c
          where c.id = ct.campaign_id
            and c.status in ('SCHEDULED', 'OPEN', 'PAUSED', 'CLOSED')
        )
    )
  );
create policy "Admins manage contestant category links"
  on public.award_contestant_categories for all
  using (public.is_admin()) with check (public.is_admin());

-- Orders: admins may read them. Nobody writes them from a browser.
create policy "Admins read vote orders"
  on public.award_vote_orders for select
  using (public.is_admin());

-- ---------------------------------------------------------------------------
-- Table privileges (see 0005: Supabase's defaults are far too broad)
-- ---------------------------------------------------------------------------
revoke all on public.award_campaigns from anon, authenticated;
revoke all on public.award_categories from anon, authenticated;
revoke all on public.award_contestants from anon, authenticated;
revoke all on public.award_contestant_categories from anon, authenticated;
revoke all on public.award_vote_orders from anon, authenticated;

-- ---------------------------------------------------------------------------
-- Results. Counts come only from PAYMENT_SUCCESS orders. Public callers get
-- them only when the campaign has show_results on; admins always do.
-- ---------------------------------------------------------------------------
create or replace function public.award_results(p_campaign_id uuid)
returns table (category_id uuid, contestant_id uuid, votes bigint)
language plpgsql
security definer
set search_path = public
stable
as $$
begin
  if not (
    public.is_admin()
    or exists (
      select 1 from public.award_campaigns c
      where c.id = p_campaign_id
        and c.show_results
        and c.status in ('OPEN', 'PAUSED', 'CLOSED')
    )
  ) then
    raise exception 'Results are not available for this campaign' using errcode = '42501';
  end if;

  return query
    select o.category_id, o.contestant_id, sum(o.quantity)::bigint
    from public.award_vote_orders o
    join public.award_contestants ct on ct.id = o.contestant_id and ct.status = 'ACTIVE'
    where o.campaign_id = p_campaign_id and o.status = 'PAYMENT_SUCCESS'
    group by o.category_id, o.contestant_id;
end;
$$;

-- Admin-only headline numbers for the dashboard.
create or replace function public.award_campaign_stats(p_campaign_id uuid)
returns table (
  total_orders bigint,
  paid_orders bigint,
  votes bigint,
  revenue numeric,
  simulated_votes bigint
)
language plpgsql
security definer
set search_path = public
stable
as $$
begin
  if not public.is_admin() then
    raise exception 'Admins only' using errcode = '42501';
  end if;

  return query
    select
      count(*)::bigint,
      count(*) filter (where o.status = 'PAYMENT_SUCCESS')::bigint,
      coalesce(sum(o.quantity) filter (where o.status = 'PAYMENT_SUCCESS'), 0)::bigint,
      coalesce(sum(o.amount) filter (where o.status = 'PAYMENT_SUCCESS' and not o.is_simulated), 0),
      coalesce(sum(o.quantity) filter (where o.status = 'PAYMENT_SUCCESS' and o.is_simulated), 0)::bigint
    from public.award_vote_orders o
    where o.campaign_id = p_campaign_id;
end;
$$;

revoke execute on function public.award_results(uuid) from public;
revoke execute on function public.award_campaign_stats(uuid) from public;
grant execute on function public.award_results(uuid) to anon, authenticated;
grant execute on function public.award_campaign_stats(uuid) to authenticated;

===== 0007_clear_test_votes.sql =====
-- Admin-only: remove the test-mode (PAYMENT_MODE=simulated) orders of a
-- campaign. Real, paid orders are never touched, so it is safe to run at
-- go-live and it lets a campaign that only has test votes be deleted.
create or replace function public.award_clear_test_votes(p_campaign_id uuid)
returns bigint
language plpgsql
security definer
set search_path = public
as $$
declare
  removed bigint;
begin
  if not public.is_admin() then
    raise exception 'Admins only' using errcode = '42501';
  end if;

  -- Test-mode orders only: ones confirmed in simulated mode, plus unpaid ones
  -- created with a sim_ reference. Real orders (gv_ references, is_simulated
  -- false) are never matched, whatever their status.
  delete from public.award_vote_orders
  where campaign_id = p_campaign_id
    and (is_simulated or payment_reference like 'sim\_%');
  get diagnostics removed = row_count;
  return removed;
end;
$$;

revoke execute on function public.award_clear_test_votes(uuid) from public;
grant execute on function public.award_clear_test_votes(uuid) to authenticated;
